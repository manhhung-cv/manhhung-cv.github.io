// Gọi TTS từ Trình duyệt và phát ngay lập tức
async function playSpeech(text, voice = "vi-VN-HoaiMyNeural", rate = "0%") {
  const res = await fetch("https://be-api-service.vercel.app/api/tts", {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      "X-App-Key": "TTS-Hunq"
    },
    body: JSON.stringify({ text, voice, rate })
  });

  if (!res.ok) throw new Error("TTS Request failed");
  const blob = await res.blob();
  const audio = new Audio(URL.createObjectURL(blob));
  audio.play();
}

playSpeech("Xin chào! Tôi là giọng đọc Edge TTS tiếng Việt.");

const axios = require('axios');
const fs = require('fs');

async function synthesize() {
  const response = await axios.post(
    'https://be-api-service.vercel.app/api/tts',
    { text: 'Xin chào từ Node.js', voice: 'vi-VN-NamMinhNeural', rate: '0%' },
    {
      headers: { 'Content-Type': 'application/json', 'X-App-Key': 'TTS-Hunq' },
      responseType: 'stream'
    }
  );
  response.data.pipe(fs.createWriteStream('speech.mp3'));
}
synthesize();
